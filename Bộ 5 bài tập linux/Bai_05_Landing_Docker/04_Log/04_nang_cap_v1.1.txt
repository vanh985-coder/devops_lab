# Quy trinh nang cap v1.0 -> v1.1 (sinh vien cung cap; khong co anh chup terminal)

nano index.html                      # sua tieu de: DevOpsMaster -> DevOpsMaster-V1
docker build -t devops-landing:v1.1 .
docker ps                            # xac dinh container dang chay truoc khi dung
docker stop landing-public
docker rm landing-public
docker run -d -p 8080:8080 \
  --restart unless-stopped \
  --name landing-public \
  --network landing-net \
  devops-landing:v1.1
